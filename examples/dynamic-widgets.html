const planButtons = document.querySelectorAll('.storage-compare .select-plan');
const planCards = document.querySelectorAll('.storage-compare .plan-card');
const status = document.querySelector('#selection-status');

planButtons.forEach((button) => {
  button.addEventListener('click', () => {
    const choice = button.dataset.choice;
    planCards.forEach((card) => {
      const selected = card.dataset.plan === choice;
      card.classList.toggle('is-selected', selected);
      card.querySelector('.select-plan').setAttribute('aria-pressed', String(selected));
    });
    status.textContent = choice === 'local'
      ? 'Local storage selected: 10 GB, offline access, and no monthly cost.'
      : 'Cloud storage selected: 100 GB, device syncing, and $8 per month.';
  });
});

(() => {
  const radios = Array.from(document.querySelectorAll('input[name="rollout"]'));
  const reasoning = document.getElementById('reasoning');
  const checkpoint = document.getElementById('checkpoint');
  const saveButton = document.getElementById('save-decision');
  const saveHint = document.getElementById('save-hint');
  const record = document.getElementById('saved-record');
  const savedChoice = document.getElementById('saved-choice');
  const savedDetails = document.getElementById('saved-details');
  const tradeoffHint = document.getElementById('tradeoff-hint');
  const editButton = document.getElementById('edit-decision');
  const tradeoffs = {
    '5% today': { hint: 'A smaller first release trades reach for a more limited initial exposure.', exposure: 'Begins with a smaller share of users.', timing: 'Starts today; broader availability can come later.', review: 'Creates a chance to observe before expanding.' },
    'Everyone today': { hint: 'A full release trades a faster, wider launch for broader initial exposure.', exposure: 'The update reaches all users at once.', timing: 'Fastest path to full availability.', review: 'Less opportunity to assess before broad release.' },
    'Postpone for review': { hint: 'More review trades speed for additional time before a release decision.', exposure: 'No new users encounter the update today.', timing: 'Availability is delayed.', review: 'Provides more time to assess before choosing a path.' }
  };
  function selectedValue() { const selected = radios.find(radio => radio.checked); return selected ? selected.value : ''; }
  function refresh() {
    const value = selectedValue();
    saveButton.disabled = !value;
    saveHint.textContent = value ? 'Add context if useful, then save your decision.' : 'Choose a rollout path to continue.';
    if (value) {
      const data = tradeoffs[value];
      tradeoffHint.textContent = data.hint;
      const descriptions = document.querySelectorAll('.tradeoff-item div span');
      descriptions[0].textContent = data.exposure;
      descriptions[1].textContent = data.timing;
      descriptions[2].textContent = data.review;
    } else {
      tradeoffHint.textContent = 'Select an option to see its considerations.';
      const defaults = ['How many users encounter the update right away?', 'How quickly does the update reach users?', 'How much time remains to assess before release?'];
      document.querySelectorAll('.tradeoff-item div span').forEach((item, index) => item.textContent = defaults[index]);
    }
  }
  radios.forEach(radio => radio.addEventListener('change', refresh));
  saveButton.addEventListener('click', () => {
    const value = selectedValue();
    if (!value) return;
    const reason = reasoning.value.trim();
    const date = checkpoint.value;
    savedChoice.textContent = value;
    const details = [];
    if (reason) details.push(reason);
    if (date) details.push(`Revisit on ${new Date(`${date}T12:00:00`).toLocaleDateString(undefined, { year: 'numeric', month: 'long', day: 'numeric' })}.`);
    if (!details.length) details.push('No additional reasoning or revisit date recorded.');
    savedDetails.textContent = details.join(' ');
    record.hidden = false;
    saveHint.textContent = 'Saved for this session. Edit any time to revisit.';
    saveButton.textContent = 'Update decision →';
  });
  editButton.addEventListener('click', () => {
    record.hidden = true;
    document.getElementById('record-heading').scrollIntoView({ behavior: 'smooth', block: 'center' });
    reasoning.focus({ preventScroll: true });
  });
  refresh();
})();
